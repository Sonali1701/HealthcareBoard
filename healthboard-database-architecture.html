Users, OAuth accounts, Sessions
-- Users table (core identity)
CREATE TABLE users (
    user_id          UUID              PRIMARY KEY DEFAULT gen_random_uuid(),
    email            VARCHAR(255)       NOT NULL UNIQUE,
    password_hash    TEXT,
    role             user_role          NOT NULL DEFAULT 'job_seeker',
    status           user_status        NOT NULL DEFAULT 'pending_verify',
    email_verified_at TIMESTAMPTZ,
    mfa_enabled      BOOLEAN           NOT NULL DEFAULT FALSE,
    mfa_secret       TEXT,
    last_login_at    TIMESTAMPTZ,
    created_at       TIMESTAMPTZ       NOT NULL DEFAULT NOW(),
    updated_at       TIMESTAMPTZ       NOT NULL DEFAULT NOW(),
    deleted_at       TIMESTAMPTZ
);

-- OAuth linked accounts
CREATE TABLE oauth_accounts (
    oauth_id          UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id           UUID         NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    provider          VARCHAR(50)  NOT NULL,
    provider_user_id  VARCHAR(255) NOT NULL,
    access_token      TEXT         NOT NULL,
    refresh_token     TEXT,
    token_expires_at  TIMESTAMPTZ  NOT NULL,
    created_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    UNIQUE(provider, provider_user_id)
);

-- Session tokens (refresh token storage)
CREATE TABLE sessions (
    session_id          UUID          PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id             UUID          NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    refresh_token_hash  TEXT          NOT NULL UNIQUE,
    device_info         JSONB,
    ip_address          INET          NOT NULL,
    expires_at          TIMESTAMPTZ   NOT NULL,
    revoked_at          TIMESTAMPTZ,
    created_at          TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);

-- Indexes for auth performance
CREATE INDEX idx_users_email     ON users(lower(email));
CREATE INDEX idx_users_role      ON users(role, status);
CREATE INDEX idx_users_last_login ON users(last_login_at DESC);
CREATE INDEX idx_sessions_expires ON sessions(expires_at)
    WHERE revoked_at IS NULL;
CREATE INDEX idx_sessions_user   ON sessions(user_id, expires_at);
Healthcare profiles, licenses, work history, certifications
-- Core profiles table (2M+ records)
CREATE TABLE profiles (
    profile_id        UUID              PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id           UUID              REFERENCES users(user_id) ON DELETE SET NULL,
    first_name        VARCHAR(100)      NOT NULL,
    last_name         VARCHAR(100)      NOT NULL,
    headline          VARCHAR(300),
    bio               TEXT,
    specialty         VARCHAR(100),
    profession_type   VARCHAR(50),
    years_experience  SMALLINT          CHECK(years_experience >= 0 AND years_experience <= 60),
    city              VARCHAR(100),
    state_code        CHAR(2),
    coordinates       GEOGRAPHY(POINT, 4326),
    open_to_work      BOOLEAN           NOT NULL DEFAULT FALSE,
    job_type_prefs    TEXT[],
    pay_min_hourly    NUMERIC(8,2)      CHECK(pay_min_hourly >= 0),
    available_date    DATE,
    npi_number        VARCHAR(10)       UNIQUE,
    profile_photo_url TEXT,
    resume_url        TEXT,
    completion_score  SMALLINT          NOT NULL DEFAULT 0
                                         CHECK(completion_score BETWEEN 0 AND 100),
    source            VARCHAR(50)       NOT NULL DEFAULT 'signup',
    search_vector     TSVECTOR          GENERATED ALWAYS AS (
        to_tsvector('english',
            coalesce(first_name,'') || ' ' ||
            coalesce(last_name,'') || ' ' ||
            coalesce(specialty,'') || ' ' ||
            coalesce(profession_type,'') || ' ' ||
            coalesce(bio,'')
        )) STORED,
    created_at        TIMESTAMPTZ       NOT NULL DEFAULT NOW(),
    updated_at        TIMESTAMPTZ       NOT NULL DEFAULT NOW()
);

-- Licenses (avg 2 licenses per profile)
CREATE TABLE licenses (
    license_id          UUID           PRIMARY KEY DEFAULT gen_random_uuid(),
    profile_id          UUID           NOT NULL REFERENCES profiles(profile_id) ON DELETE CASCADE,
    license_type        VARCHAR(50)    NOT NULL,
    license_number      VARCHAR(100)   NOT NULL,
    state_code          CHAR(2)        NOT NULL,
    status              license_status  NOT NULL DEFAULT 'active',
    issued_date         DATE,
    expiry_date         DATE,
    verified_at         TIMESTAMPTZ,
    verification_source VARCHAR(100),
    is_compact          BOOLEAN        NOT NULL DEFAULT FALSE,
    created_at          TIMESTAMPTZ    NOT NULL DEFAULT NOW()
);

-- Work history records
CREATE TABLE work_history (
    work_id           UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
    profile_id        UUID         NOT NULL REFERENCES profiles(profile_id) ON DELETE CASCADE,
    employer_name     VARCHAR(200) NOT NULL,
    job_title         VARCHAR(200) NOT NULL,
    specialty         VARCHAR(100),
    employment_type   VARCHAR(50),
    start_date        DATE         NOT NULL,
    end_date          DATE,
    city              VARCHAR(100),
    state_code        CHAR(2),
    description       TEXT,
    bed_count         SMALLINT,
    nurse_ratio       VARCHAR(20),
    CHECK(end_date IS NULL OR end_date >= start_date)
);

-- ── GEO RADIUS SEARCH FUNCTION ──
CREATE OR REPLACE FUNCTION jobs_within_radius(
    lat      FLOAT,
    lng      FLOAT,
    miles    FLOAT
) RETURNS SETOF job_postings AS $$
    SELECT * FROM job_postings
    WHERE status = 'active'
    AND ST_DWithin(
        coordinates,
        ST_SetSRID(ST_MakePoint(lng, lat), 4326),
        miles * 1609.34  -- convert miles to meters
    )
    ORDER BY ST_Distance(
        coordinates,
        ST_SetSRID(ST_MakePoint(lng, lat), 4326)
    );
$$ LANGUAGE sql STABLE;
Automation triggers and utility functions
-- Auto-update updated_at on any row change
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_profiles_updated_at
    BEFORE UPDATE ON profiles
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_jobs_updated_at
    BEFORE UPDATE ON job_postings
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- Increment application_count on job when application inserted
CREATE OR REPLACE FUNCTION increment_application_count()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE job_postings
    SET application_count = application_count + 1
    WHERE job_id = NEW.job_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_inc_app_count
    AFTER INSERT ON applications
    FOR EACH ROW EXECUTE FUNCTION increment_application_count();

-- Auto-expire jobs past their expires_at (run via pg_cron hourly)
CREATE OR REPLACE FUNCTION expire_old_jobs()
RETURNS INTEGER AS $$
DECLARE
    expired_count INTEGER;
BEGIN
    UPDATE job_postings
    SET status = 'expired'
    WHERE status = 'active'
      AND expires_at < NOW();
    GET DIAGNOSTICS expired_count = ROW_COUNT;
    RETURN expired_count;
END;
$$ LANGUAGE plpgsql;

-- Profile completion score calculator
CREATE OR REPLACE FUNCTION calculate_completion_score(p_profile_id UUID)
RETURNS SMALLINT AS $$
DECLARE
    score INTEGER := 0;
    p     profiles%ROWTYPE;
BEGIN
    SELECT * INTO p FROM profiles WHERE profile_id = p_profile_id;
    IF p.headline    IS NOT NULL THEN score := score + 10; END IF;
    IF p.bio         IS NOT NULL THEN score := score + 10; END IF;
    IF p.specialty   IS NOT NULL THEN score := score + 15; END IF;
    IF p.city        IS NOT NULL THEN score := score + 10; END IF;
    IF p.resume_url  IS NOT NULL THEN score := score + 20; END IF;
    IF p.profile_photo_url IS NOT NULL THEN score := score + 10; END IF;
    -- Check for at least one license
    IF EXISTS(SELECT 1 FROM licenses
              WHERE profile_id = p_profile_id
              AND status = 'active') THEN
        score := score + 15;
    END IF;
    -- Check for work history
    IF EXISTS(SELECT 1 FROM work_history
              WHERE profile_id = p_profile_id) THEN
        score := score + 10;
    END IF;
    RETURN LEAST(score, 100);
END;
$$ LANGUAGE plpgsql;

🔔 notifications — Range Partitioned by created_at (monthly)
Notifications grow at ~3M rows/month. Monthly partitions allow instant deletion of read notifications older than 6 months without expensive DELETE operations — just DROP TABLE notifications_2024_06.
Retention policy: keep unread forever, delete read notifications after 90 days via partitioned cleanup job.
📊 audit_logs — Range Partitioned by created_at (HIPAA: 7-year retention)
HIPAA requires audit logs be retained for 6 years (often 7 for safe margin). Partitioning by month allows efficient archival — older partitions are moved to S3 Glacier after 1 year using pg_partman + custom archival scripts. Hot partitions stay on SSD, cold on cheaper storage.
🌍 Read Replica Architecture
⚡
Primary (Write)
All INSERTs, UPDATEs, DELETEs. Multi-AZ for failover. RDS PostgreSQL 16 + 32 vCPU, 128GB RAM.
🔍
Read Replica 1 (Search)
All profile and job search SELECTs. Dedicated replica for Elasticsearch sync. Lag target: <500ms.
📊
Read Replica 2 (Analytics)
Reporting queries, employer dashboards, data warehouse exports. Can lag up to 5min.
Connection pooling via PgBouncer in transaction mode. Pools: 200 connections to primary, 500 to each read replica. Application uses read_preference routing based on query type.
